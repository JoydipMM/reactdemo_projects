Checking If a Year Is a Leap Year

function isLeapYear(year) { return (year % 4 === 0 && year % 100 !== 0) || year % 400 === 0; } document.write("2025 Is a Leap Year? "); document.write(isLeapYear(2025)); // False document.write("<br>"); document.write("2024 Is a Leap Year? "); document.write(isLeapYear(2024)); // true document.write("<br>"); document.write("2023 Is a Leap Year? "); document.write(isLeapYear(2023)); // false document.write("<br>");
