find & findIndex Method

The find() method in JavaScript returns the first element in an array that satisfies a given condition. If no element matches, it returns undefined.

  • Purpose: Finds the first element that meets the condition.
  • Return Value: Returns the first matching element or undefined if none match.
  • Stops Searching? yes, stops after finding the first match.
  • Use Case: When you need the actual element.
var age = [10, 18, 23, 8, 11]; document.write("Actual Array: "+ age + "<br>"); function varifyAge(age){ return age >= 18; } var b = age.find(varifyAge) document.write("Find Data: "+ b + "<br>"); var b2 = age.find(age => age >=18) document.write("Find Data with arrow function: "+ b2 + "<br>"); var b = age.findIndex(varifyAge) document.write("findIndex: "+ b + "<br>");
