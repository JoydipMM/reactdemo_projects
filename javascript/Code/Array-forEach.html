Array forEach Loop

The forEach() method is used to iterate over an array and execute a provided function once for each element.

var arrayData = ["Sanjay", "Aman", "Raman", "Karan"]; document.write("Actual Array: "+arrayData+ "<br>"); arrayData.forEach(function(item, index){ document.write("Index- "+index+": "+item+ "<br>"); }) document.write("<br><br>"); arrayData.forEach( item => document.write(item+ "<br>") ); document.write("<br><br>"); arrayData.forEach( (item, index) => document.write("Index- "+index+": "+item+ "<br>") );
