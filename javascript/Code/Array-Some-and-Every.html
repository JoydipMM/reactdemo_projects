Some Method

The some() method in JavaScript is used to check if at least one element in an array satisfies a given condition. It returns true if any element meets the condition; otherwise, it returns false.

  • Purpose: Checks if at least one element meets the condition.
  • Return Value: Returns true if at least one element matches, otherwise false.
  • Stops Searching? Yes, stops after finding the first match.
  • Use Case: When you only need to know if a match exists.
var age = [10, 18, 8, 11]; document.write("Actual Array: "+ age + "<br>"); function varifyAge(age){ return age >= 18; } var b = age.some( varifyAge ) document.write("Some Data: "+ b + "<br>");

Every Method

The every() method in JavaScript checks if all elements in an array satisfy a given condition. It returns true only if every element passes the test; otherwise, it returns false.

var age_1 = [10, 18, 8, 11]; document.write("Actual Array: "+ age_1 + "<br>"); function varifyEveryAge(age){ return age >= 18; } var vage = age_1.every(varifyEveryAge) document.write("Some Data: "+ vage + "<br>");
