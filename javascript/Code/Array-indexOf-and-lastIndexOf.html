indexOf

indexOf = indexOf("Search item", Start)
Note: if data not found then result will print -1

var a = ["Sanjay", "Aman", "Raman", "Karan", "Aman"]; document.write("Actual Array: "+ a + "<br>"); var b = a.indexOf("Aman"); document.write('a.indexOf("Aman"):' + b + "<br>"); var b = a.indexOf("Aman", 3); document.write('a.indexOf("Aman", 3) : '+ b + "<br>");

lastIndexOf

lastIndexOf = lastIndexOf("Search item", Start) it start from last index
Note: if data not found then result will print -1
